Abstract

Quantifying the space use of the pigs in East Alligator, and how that changes across seasons and for different months of the year. For estimating space use we are using ‘occurrence’ distributions, which are the probability of finding an animal in a given area at a given time. This is different to ‘range’ distributions, which are the probability of finding an animal in a given area over a long period of time (i.e. their home range). Occurrence distributions are more appropriate for short-term space use, such as across seasons and months.

Convert to sf and save for QGIS

as.sf() returns three polygons per pig for each contour level - the lower CI, the estimate and the upper CI. We extract the 50% (core area) and 95% (home range) contours. Each UD is in ctmm’s own local projection, so they are all transformed to EPSG:3112 to match the rest of the data.

  # one row per pig per contour level
  crossing(level_ud = contour_levels) %>%
  mutate(hr_sf = map2(ud, level_ud, ~ as.sf(.x, level.UD = .y, level = 0.95) %>%
                        st_transform(crs = 3112))) %>%
  dplyr::select(name, level_ud, hr_sf) %>%
  # as.sf already has a 'name' column (e.g. "Betty 95% est"), so rename it
  mutate(hr_sf = map(hr_sf, ~ rename(.x, contour = name))) %>%
  unnest(hr_sf) %>%
  mutate(ci = word(contour, -1)) %>%
  st_as_sf(crs = 3112)

Using ctmm Kriging estimates

For each pig-month we fit a movement model (pHREML) and then estimate the occurrence distribution with occurrence(), which uses time-series Kriging to estimate where the pig was during the sampling period. Since we already have the movement model, we also estimate the monthly AKDE home range (which estimates the range the pig would use in the long run if it continued to behave as it did that month).

The model type of each pig (e.g. OUF anisotropic) is taken from the model selected for that pig above (pigs_nested$fit), so the same type is used for every month of a pig. The parameters (e.g. the range and the positional and velocity autocorrelation time scales) are still estimated separately for each month, as movement rates differ between seasons. This is faster and more stable than running model selection for each pig-month. Pigs without a fit above fall back to model selection for each month.

The model types above should be selected from the full tracking period, i.e. with pig_data <- pig_all when the ‘All pigs’ section is run.

Note that the area of the occurrence distribution is not a home range area. It represents the uncertainty in the pig’s path, so it depends on the sampling rate (which is the same across all pigs here, so comparisons between pigs and months are still valid).

Model fitting is slow, so this runs in parallel and the results are saved and re-loaded if they already exist. Delete the file to re-fit.

Show code 
# convert one pig-month of data to a ctmm telemetry object
# (as.telemetry() identifies columns by their Movebank names, and x1_/y1_
# are in GDA94 / GA Lambert, EPSG:3112)
to_telemetry <- function(df, id) {
  df %>%
    st_as_sf(coords = c("x1_", "y1_"), crs = 3112) %>%
    st_transform(crs = 4326) %>%
    mutate(location.long = st_coordinates(.)[, 1],
           location.lat  = st_coordinates(.)[, 2]) %>%
    st_drop_geometry() %>%
    transmute(ID = id, timestamp = t1_, location.long, location.lat) %>%
    as.data.frame() %>%
    as.telemetry(timezone = "Australia/Darwin") %>%
    suppressMessages()
}

# fit the movement model and return the occurrence and AKDE contours as sf
# (in EPSG:3112). The UD rasters themselves are not kept to save memory.
# pig_fit is the model selected for that pig over the full tracking period,
# which sets the model type. The parameters are estimated from the month's data.
fit_monthly_ctmm <- function(tel, pig_fit, levels) {

  if (is.null(pig_fit)) {
    guess <- ctmm.guess(tel, interactive = FALSE)
    fit <- ctmm.select(tel, guess)
  } else {
    # the guess takes the model structure from pig_fit, and its parameters as
    # starting values (the mean location is re-estimated), which are then
    # all re-estimated from this month's data
    guess <- ctmm.guess(tel, CTMM = pig_fit, interactive = FALSE)
    fit <- ctmm.fit(tel, guess, method = "pHREML")
  }

  od <- occurrence(tel, fit)
  ud <- akde(tel, fit)

  # as.sf() has a 'name' column (e.g. "Betty 95% est")
  contours <- function(x, method) {
    lapply(levels, function(l) {
      s <- sf::st_transform(as.sf(x, level.UD = l, level = 0.95), crs = 3112)
      s$level_ud <- l
      s$method <- method
      s
    }) %>%
      do.call(rbind, .)
  }

  list(fit = fit,
       contours = rbind(contours(od, "Occurrence (Kriging)"),
                        contours(ud, "AKDE")))
}

hr_monthly_ctmm_file <- paste0(outputs_dir, "pig_monthly_ctmm.rds")

if (file.exists(hr_monthly_ctmm_file)) {

  pig_monthly_ctmm <- readRDS(hr_monthly_ctmm_file)

} else {

  pig_monthly_ctmm <- pig_monthly %>%

    { if (is.na(n_test)) . else slice_head(., n = n_test) } %>%

  pig_monthly_ctmm <- pig_monthly_ctmm %>%

      .options = furrr::furrr_options(seed = TRUE, packages = c("ctmm", "sf")),
      .progress = TRUE))
  future::plan(future::sequential)
  toc()

  pig_monthly_ctmm <- pig_monthly_ctmm %>% dplyr::select(-data, -pig_fit)
  saveRDS(pig_monthly_ctmm, hr_monthly_ctmm_file)

}

monthly_hr_map <- function(hr_sf, pts_sf, title, interval_ms = 1500) {

  hr_sf <- hr_sf %>%
    filter(ci == "est") %>%
    st_transform(crs = 4326) %>%
    mutate(popup = paste0("ID: ", name,
                          "<br>Month: ", format(month_date, "%b %Y"),
                          "<br>Contour: ", level_ud * 100, "%",
                          "<br>Area: ", round(area_km2, 2), " km<sup>2</sup>"))

  months <- sort(unique(hr_sf$year_month))
  pal <- colorFactor(viridisLite::viridis(n_distinct(pig_all$name), option = "H"),
                     domain = sort(unique(pig_all$name)))

  m <- leaflet(options = leafletOptions(preferCanvas = TRUE)) %>%
    addProviderTiles(providers$Esri.WorldImagery)

  for (mon in months) {
    hr_95 <- hr_sf %>% filter(year_month == mon, level_ud == 0.95)
    hr_50 <- hr_sf %>% filter(year_month == mon, level_ud == 0.5)
    pts <- pts_sf %>% filter(year_month == mon)

    m <- m %>%
      addCircleMarkers(data = pts, radius = 2, stroke = FALSE,
                       fillColor = ~ pal(name), fillOpacity = 0.6,
                       group = paste0(mon, "_fixes")) %>%
      addPolygons(data = hr_95, color = ~ pal(name), weight = 2, opacity = 1,
                  fillColor = ~ pal(name), fillOpacity = 0.1,
                  popup = ~ popup, group = mon) %>%
      addPolygons(data = hr_50, color = ~ pal(name), weight = 1.5, opacity = 1,
                  dashArray = "4 4",
                  fillColor = ~ pal(name), fillOpacity = 0.3,
                  popup = ~ popup, group = mon)
  }

  m %>%
    addLegend(position = "bottomright", pal = pal,
              values = sort(unique(hr_sf$name)), title = "Pig ID") %>%
    addScaleBar(position = "bottomleft",
                options = scaleBarOptions(imperial = FALSE)) %>%
    htmlwidgets::onRender(month_slider_js,
                          data = list(months = months,
                                      labels = format(ym(months), "%B %Y"),
                                      title = title,
                                      interval = interval_ms))
}

## Using ctmm Kriging estimates

For each pig-month we fit a movement model (pHREML) and then estimate the occurrence distribution with `occurrence()`, which uses time-series Kriging to estimate where the pig was during the sampling period. Since we already have the movement model, we also estimate the monthly AKDE home range (which estimates the range the pig would use in the long run if it continued to behave as it did that month).

The model type of each pig (e.g. OUF anisotropic) is taken from the model selected for that pig above (`pigs_nested$fit`), so the same type is used for every month of a pig. The parameters (e.g. the range and the positional and velocity autocorrelation time scales) are still estimated separately for each month, as movement rates differ between seasons. This is faster and more stable than running model selection for each pig-month. Pigs without a fit above fall back to model selection for each month.

The model types above should be selected from the full tracking period, i.e. with `pig_data <- pig_all` when the 'All pigs' section is run.

Note that the area of the occurrence distribution is not a home range area. It represents the uncertainty in the pig's path, so it depends on the sampling rate (which is the same across all pigs here, so comparisons between pigs and months are still valid).

Model fitting is slow, so this runs in parallel and the results are saved and re-loaded if they already exist. Delete the file to re-fit.

```{r}
# convert one pig-month of data to a ctmm telemetry object
# (as.telemetry() identifies columns by their Movebank names, and x1_/y1_
# are in GDA94 / GA Lambert, EPSG:3112)
to_telemetry <- function(df, id) {
  df %>%
    st_as_sf(coords = c("x1_", "y1_"), crs = 3112) %>%
    st_transform(crs = 4326) %>%
    mutate(location.long = st_coordinates(.)[, 1],
           location.lat  = st_coordinates(.)[, 2]) %>%
    st_drop_geometry() %>%
    transmute(ID = id, timestamp = t1_, location.long, location.lat) %>%
    as.data.frame() %>%
    as.telemetry(timezone = "Australia/Darwin") %>%
    suppressMessages()
}

# fit the movement model and return the occurrence and AKDE contours as sf
# (in EPSG:3112). The UD rasters themselves are not kept to save memory.
# pig_fit is the model selected for that pig over the full tracking period,
# which sets the model type. The parameters are estimated from the month's data.
fit_monthly_ctmm <- function(tel, pig_fit, levels) {

  if (is.null(pig_fit)) {
    guess <- ctmm.guess(tel, interactive = FALSE)
    fit <- ctmm.select(tel, guess)
  } else {
    # the guess takes the model structure from pig_fit, and its parameters as
    # starting values (the mean location is re-estimated), which are then
    # all re-estimated from this month's data
    guess <- ctmm.guess(tel, CTMM = pig_fit, interactive = FALSE)
    fit <- ctmm.fit(tel, guess, method = "pHREML")
  }

  od <- occurrence(tel, fit)
  ud <- akde(tel, fit)

  # as.sf() has a 'name' column (e.g. "Betty 95% est")
  contours <- function(x, method) {
    lapply(levels, function(l) {
      s <- sf::st_transform(as.sf(x, level.UD = l, level = 0.95), crs = 3112)
      s$level_ud <- l
      s$method <- method
      s
    }) %>%
      do.call(rbind, .)
  }

  list(fit = fit,
       contours = rbind(contours(od, "Occurrence (Kriging)"),
                        contours(ud, "AKDE")))
}

hr_monthly_ctmm_file <- paste0(outputs_dir, "pig_monthly_ctmm.rds")

if (file.exists(hr_monthly_ctmm_file)) {

  pig_monthly_ctmm <- readRDS(hr_monthly_ctmm_file)

} else {

  pig_monthly_ctmm <- pig_monthly %>%
    dplyr::select(name, year_month, month_date, data) %>%
    { if (is.na(n_test)) . else slice_head(., n = n_test) } %>%
    mutate(telemetry = map2(data, name, to_telemetry)) %>%
    # model selected for each pig above (NULL for pigs without one)
    left_join(pigs_nested %>% dplyr::select(name, pig_fit = fit), by = "name")

  tic()
  future::plan(future::multisession, workers = max(1, parallelly::availableCores() - 2))
  pig_monthly_ctmm <- pig_monthly_ctmm %>%
    mutate(ctmm = furrr::future_map2(
      telemetry, pig_fit,
      # a failed fit returns NULL rather than stopping the whole loop
      possibly(~ fit_monthly_ctmm(.x, .y, contour_levels), otherwise = NULL),
      .options = furrr::furrr_options(seed = TRUE, packages = c("ctmm", "sf")),
      .progress = TRUE))
  future::plan(future::sequential)
  toc()

  pig_monthly_ctmm <- pig_monthly_ctmm %>% dplyr::select(-data, -pig_fit)
  saveRDS(pig_monthly_ctmm, hr_monthly_ctmm_file)
  beep()

}

# any pig-months where the fit failed
pig_monthly_ctmm %>% filter(map_lgl(ctmm, is.null)) %>% dplyr::select(name, year_month)

# movement models fitted for each pig (should be one type per pig, unless
```
