Abstract

Quantifying the space use of the pigs in East Alligator, and how that changes across seasons and for different months of the year. For estimating space use we are using ‘occurrence’ distributions, which are the probability of finding an animal in a given area at a given time. This is different to ‘range’ distributions, which are the probability of finding an animal in a given area over a long period of time (i.e. their home range). Occurrence distributions are more appropriate for short-term space use, such as across seasons and months.

as.sf() returns three polygons per pig for each contour level - the lower CI, the estimate and the upper CI. We extract the 50% (core area) and 95% (home range) contours. Each UD is in ctmm’s own local projection, so they are all transformed to EPSG:3112 to match the rest of the data.

AKDE seasonal

Here we estimate weighted AKDE home ranges separately for each season. Each season is kept separate by year (e.g. the 2025 dry season and the 2026 dry season), and the wet season spans the new year (e.g. the 2025-26 wet season).

The model type of each pig (e.g. OUF anisotropic) is taken from the model selected for its full tracking period (pigs_nested$fit), so the same type is used for every season of a pig. The parameters are then re-estimated for each season. This matters because AKDE takes its effective sample size (and therefore its CIs) from the movement model, so using the full-period model directly would give each season the certainty of the whole tracking period, and CIs that are too narrow.

The seasons are the same as in the data preparation and weather scripts:

  • Wet season: November - April
    • Early wet: November - January
    • Late wet: February - April
  • Dry season: May - October
    • Early dry: May - July
    • Late dry: August - October

# season of each timestamp, labelled by year
season_id <- function(t) {
  m <- month(t)
  y <- year(t)
  case_when(
    m %in% 5:10 ~ paste(y, "dry"),
    m %in% 11:12 ~ paste0(y, "-", sprintf("%02d", (y + 1) %% 100), " wet"),
    m %in% 1:4 ~ paste0(y - 1, "-", sprintf("%02d", y %% 100), " wet")
  )
}

# sub-season of each timestamp, labelled by year
subseason_id <- function(t) {
  m <- month(t)
  y <- year(t)
  case_when(
    m %in% 5:7 ~ paste(y, "early dry"),
    m %in% 8:10 ~ paste(y, "late dry"),
    m %in% 11:12 ~ paste0(y, "-", sprintf("%02d", (y + 1) %% 100), " early wet"),
    m == 1 ~ paste0(y - 1, "-", sprintf("%02d", y %% 100), " early wet"),
    m %in% 2:4 ~ paste(y, "late wet")
  )
}

# split one pig's telemetry object into periods (seasons or sub-seasons).
# Subsetting the telemetry object keeps its projection, so it matches the
# full-period movement model.
split_telemetry <- function(tel, period_fun) {
  t <- with_tz(tel$timestamp, tzone = "Australia/Darwin")
  periods <- period_fun(t)
  tibble(period = unique(periods)) %>%
    mutate(telemetry = map(period, ~ tel[periods == .x, ]),
           start = map_vec(period, ~ min(t[periods == .x])),
           n_fixes = map_int(telemetry, nrow),
           n_days = map_int(period, ~ n_distinct(as_date(t[periods == .x]))))
}

# fit the movement model to one pig-period and estimate the weighted AKDE.
# The guess takes the model type from pig_fit (the full-period model), and its
# parameters as starting values, which are then re-estimated from the period's data
fit_period_akde <- function(tel, pig_fit) {
  guess <- ctmm.guess(tel, CTMM = pig_fit, interactive = FALSE)
  fit <- ctmm.fit(tel, guess, method = "pHREML")
  list(fit = fit, ud = akde(tel, fit, weights = TRUE))
}

# one row per pig per period, with the period's movement model and weighted
# AKDE. This runs in parallel, and the results are saved and re-loaded if they
# already exist (delete the file to re-run).
period_akde <- function(pigs_nested, period_fun, min_days, file) {

  if (file.exists(file)) return(readRDS(file))

  pig_periods <- pigs_nested %>%
    dplyr::select(name, telemetry, pig_fit = fit) %>%
    mutate(periods = map(telemetry, ~ split_telemetry(.x, period_fun))) %>%
    dplyr::select(-telemetry) %>%
    unnest(periods) %>%
    filter(n_days >= min_days)

  tic()
  future::plan(future::multisession, workers = max(1, parallelly::availableCores() - 2))
  results <- furrr::future_map2(
    pig_periods$telemetry, pig_periods$pig_fit,
    # a failed fit returns NULL rather than stopping the whole loop
    possibly(fit_period_akde, otherwise = NULL),
    .options = furrr::furrr_options(seed = TRUE, packages = "ctmm"),
    .progress = TRUE)
  future::plan(future::sequential)
  toc()

  pig_periods <- pig_periods %>%
    mutate(fit = map(results, "fit"),
           ud = map(results, "ud")) %>%
    dplyr::select(-pig_fit)

  # any pig-periods where the fit failed
  failed <- pig_periods %>% filter(map_lgl(ud, is.null))
  if (nrow(failed) > 0) {
    warning("Fit failed for: ", paste(failed$name, failed$period, collapse = ", "))
  }
  pig_periods <- pig_periods %>% filter(!map_lgl(ud, is.null))

  saveRDS(pig_periods, file)
  beep(sound = 2)
  pig_periods
}

# the number of days of each pig in each period, including those that
# are removed for having fewer than min_days
period_days <- function(pigs_nested, period_fun) {
  pigs_nested %>%
    mutate(periods = map(telemetry, ~ split_telemetry(.x, period_fun))) %>%
    dplyr::select(name, periods) %>%
    unnest(periods) %>%
    dplyr::select(name, period, start, n_fixes, n_days)
}

# 50% and 95% contours (with CIs) of each pig-period as sf (in EPSG:3112)
period_contours <- function(pig_periods, levels = contour_levels) {

  # periods ordered by when they start
  period_levels <- pig_periods %>%
    summarise(start = min(start), .by = period) %>%
    arrange(start) %>%
    pull(period)

  pig_periods %>%
    dplyr::select(name, period, ud) %>%
    crossing(level_ud = levels) %>%
    mutate(hr_sf = map2(ud, level_ud, ~ as.sf(.x, level.UD = .y, level = 0.95) %>%
                          st_transform(crs = 3112))) %>%
    dplyr::select(name, period, level_ud, hr_sf) %>%
    # as.sf already has a 'name' column (e.g. "Betty 95% est"), so rename it
    mutate(hr_sf = map(hr_sf, ~ rename(.x, contour = name))) %>%
    unnest(hr_sf) %>%
    st_as_sf(crs = 3112) %>%
    mutate(ci = word(contour, -1),
           period = factor(period, levels = period_levels),
           area_km2 = as.numeric(st_area(.)) / 1e6) %>%
    dplyr::select(name, period, level_ud, ci, area_km2)
}

# area of each pig in each period, with 95% CIs
plot_period_areas <- function(hr_sf, title) {

  areas <- hr_sf %>%
    st_drop_geometry() %>%
    pivot_wider(names_from = ci, values_from = area_km2) %>%
    mutate(level = paste0(level_ud * 100, "%"))

  dodge <- position_dodge(width = 0.6)

  ggplot(areas, aes(x = period, y = est, colour = name, group = name)) +
    geom_line(position = dodge, alpha = 0.3, linewidth = 0.3) +
    geom_errorbar(aes(ymin = low, ymax = high),
                  position = dodge, width = 0, alpha = 0.5) +
    geom_point(position = dodge, size = 1) +
    # mean across pigs
    stat_summary(aes(group = 1), fun = mean, geom = "point",
                 colour = "black", shape = 4, size = 3, stroke = 1) +
    scale_colour_viridis_d(guide = "none") +
    scale_y_continuous(expression("Area (km"^2*")")) +
    facet_wrap(~ level, scales = "free_y", ncol = 1) +
    labs(title = title,
         subtitle = "Weighted AKDE with 95% CIs. Crosses are the mean across pigs",
         x = "") +
    theme_bw()
}

# leaflet map of the contour point estimates, with radio buttons to choose
# the period, showing that period's (thinned) GPS fixes
period_hr_map <- function(hr_sf, pts_sf, title) {

  hr_sf <- hr_sf %>%
    filter(ci == "est") %>%
    st_transform(crs = 4326) %>%
    mutate(popup = paste0("ID: ", name,
                          "<br>Period: ", period,
                          "<br>Contour: ", level_ud * 100, "%",
                          "<br>Area: ", round(area_km2, 2), " km<sup>2</sup>"))

  periods <- levels(droplevels(hr_sf$period))

  # only show the fixes of pig-periods with a home range (i.e. not those
  # removed for having too few days)
  pts_sf <- pts_sf %>%
    semi_join(hr_sf %>% st_drop_geometry() %>%
                distinct(name, period) %>%
                mutate(period = as.character(period)),
              by = c("name", "period"))

  pal <- colorFactor(viridisLite::viridis(n_distinct(pig_all$name), option = "H"),
                     domain = sort(unique(pig_all$name)))

  m <- leaflet(options = leafletOptions(preferCanvas = TRUE)) %>%
    addProviderTiles(providers$Esri.WorldImagery)

  # each period is a 'base group', so the layers control shows radio buttons
  for (p in periods) {
    m <- m %>%
      addCircleMarkers(data = pts_sf %>% filter(period == p),
                       radius = 2, stroke = FALSE,
                       fillColor = ~ pal(name), fillOpacity = 0.5,
                       group = p) %>%
      addPolygons(data = hr_sf %>% filter(period == p, level_ud == 0.95),
                  color = ~ pal(name), weight = 2, opacity = 1,
                  fillColor = ~ pal(name), fillOpacity = 0.1,
                  popup = ~ popup, group = p) %>%
      addPolygons(data = hr_sf %>% filter(period == p, level_ud == 0.5),
                  color = ~ pal(name), weight = 1.5, opacity = 1,
                  dashArray = "4 4",
                  fillColor = ~ pal(name), fillOpacity = 0.3,
                  popup = ~ popup, group = p)
  }

  m %>%
    addLegend(position = "bottomright", pal = pal,
              values = sort(unique(hr_sf$name)), title = "Pig ID") %>%
    addScaleBar(position = "bottomleft",
                options = scaleBarOptions(imperial = FALSE)) %>%
    addControl(paste0("<b>", title, "</b>"), position = "topleft") %>%
    addLayersControl(baseGroups = periods,
                     options = layersControlOptions(collapsed = FALSE))
}

# map of monthly contours (point estimates) with a month slider
monthly_hr_map <- function(hr_sf, pts_sf, title, interval_ms = 1500) {

  hr_sf <- hr_sf %>%
    # filter(ci == "est") %>%
    st_transform(crs = 4326) %>%
    mutate(popup = paste0("ID: ", name,
                          "<br>Month: ", format(month_date, "%b %Y"),
                          "<br>Contour: ", level_ud * 100, "%",
                          "<br>Area: ", round(area_km2, 2), " km<sup>2</sup>"))

  months <- sort(unique(hr_sf$year_month))
  pal <- colorFactor(viridisLite::viridis(n_distinct(pig_all$name), option = "H"),
                     domain = sort(unique(pig_all$name)))

  m <- leaflet(options = leafletOptions(preferCanvas = TRUE)) %>%
    addProviderTiles(providers$Esri.WorldImagery)

  for (mon in months) {
    hr_95 <- hr_sf %>% filter(year_month == mon, level_ud == 0.95)
    hr_50 <- hr_sf %>% filter(year_month == mon, level_ud == 0.5)
    pts <- pts_sf %>% filter(year_month == mon)

    m <- m %>%
      addCircleMarkers(data = pts, radius = 2, stroke = FALSE,
                       fillColor = ~ pal(name), fillOpacity = 0.6,
                       group = paste0(mon, "_fixes")) %>%
      addPolygons(data = hr_95, color = ~ pal(name), weight = 2, opacity = 1,
                  fillColor = ~ pal(name), fillOpacity = 0.1,
                  popup = ~ popup, group = mon) %>%
      addPolygons(data = hr_50, color = ~ pal(name), weight = 1.5, opacity = 1,
                  dashArray = "4 4",
                  fillColor = ~ pal(name), fillOpacity = 0.3,
                  popup = ~ popup, group = mon)
  }

  m %>%
    addLegend(position = "bottomright", pal = pal,
              values = sort(unique(hr_sf$name)), title = "Pig ID") %>%
    addScaleBar(position = "bottomleft",
                options = scaleBarOptions(imperial = FALSE)) %>%
    htmlwidgets::onRender(month_slider_js,
                          data = list(months = months,
                                      labels = format(ym(months), "%B %Y"),
                                      title = title,
                                      interval = interval_ms))
}

# AKDE seasonal

Here we estimate weighted AKDE home ranges separately for each season. Each season is kept separate by year (e.g. the 2025 dry season and the 2026 dry season), and the wet season spans the new year (e.g. the 2025-26 wet season).

The model type of each pig (e.g. OUF anisotropic) is taken from the model selected for its full tracking period (`pigs_nested$fit`), so the same type is used for every season of a pig. The parameters are then re-estimated for each season. This matters because AKDE takes its effective sample size (and therefore its CIs) from the movement model, so using the full-period model directly would give each season the certainty of the whole tracking period, and CIs that are too narrow.

The seasons are the same as in the data preparation and weather scripts:

- Wet season: November - April
  - Early wet: November - January
  - Late wet: February - April
- Dry season: May - October
  - Early dry: May - July
  - Late dry: August - October

# season of each timestamp, labelled by year
season_id <- function(t) {
  m <- month(t)
  y <- year(t)
  case_when(
    m %in% 5:10 ~ paste(y, "dry"),
    m %in% 11:12 ~ paste0(y, "-", sprintf("%02d", (y + 1) %% 100), " wet"),
    m %in% 1:4 ~ paste0(y - 1, "-", sprintf("%02d", y %% 100), " wet")
  )
}

# sub-season of each timestamp, labelled by year
subseason_id <- function(t) {
  m <- month(t)
  y <- year(t)
  case_when(
    m %in% 5:7 ~ paste(y, "early dry"),
    m %in% 8:10 ~ paste(y, "late dry"),
    m %in% 11:12 ~ paste0(y, "-", sprintf("%02d", (y + 1) %% 100), " early wet"),
    m == 1 ~ paste0(y - 1, "-", sprintf("%02d", y %% 100), " early wet"),
    m %in% 2:4 ~ paste(y, "late wet")
  )
}

# split one pig's telemetry object into periods (seasons or sub-seasons).
# Subsetting the telemetry object keeps its projection, so it matches the
# full-period movement model.
split_telemetry <- function(tel, period_fun) {
  t <- with_tz(tel$timestamp, tzone = "Australia/Darwin")
  periods <- period_fun(t)
  tibble(period = unique(periods)) %>%
    mutate(telemetry = map(period, ~ tel[periods == .x, ]),
           start = map_vec(period, ~ min(t[periods == .x])),
           n_fixes = map_int(telemetry, nrow),
           n_days = map_int(period, ~ n_distinct(as_date(t[periods == .x]))))
}

# fit the movement model to one pig-period and estimate the weighted AKDE.
# The guess takes the model type from pig_fit (the full-period model), and its
# parameters as starting values, which are then re-estimated from the period's data
fit_period_akde <- function(tel, pig_fit) {
  guess <- ctmm.guess(tel, CTMM = pig_fit, interactive = FALSE)
  fit <- ctmm.fit(tel, guess, method = "pHREML")
  list(fit = fit, ud = akde(tel, fit, weights = TRUE))
}

# one row per pig per period, with the period's movement model and weighted
# AKDE. This runs in parallel, and the results are saved and re-loaded if they
# already exist (delete the file to re-run).
period_akde <- function(pigs_nested, period_fun, min_days, file) {

  if (file.exists(file)) return(readRDS(file))

  pig_periods <- pigs_nested %>%
    dplyr::select(name, telemetry, pig_fit = fit) %>%
    mutate(periods = map(telemetry, ~ split_telemetry(.x, period_fun))) %>%
    dplyr::select(-telemetry) %>%
    unnest(periods) %>%
    filter(n_days >= min_days)

  tic()
  future::plan(future::multisession, workers = max(1, parallelly::availableCores() - 2))
  results <- furrr::future_map2(
    pig_periods$telemetry, pig_periods$pig_fit,
    # a failed fit returns NULL rather than stopping the whole loop
    possibly(fit_period_akde, otherwise = NULL),
    .options = furrr::furrr_options(seed = TRUE, packages = "ctmm"),
    .progress = TRUE)
  future::plan(future::sequential)
  toc()

  pig_periods <- pig_periods %>%
    mutate(fit = map(results, "fit"),
           ud = map(results, "ud")) %>%
    dplyr::select(-pig_fit)

  # any pig-periods where the fit failed
  failed <- pig_periods %>% filter(map_lgl(ud, is.null))
  if (nrow(failed) > 0) {
    warning("Fit failed for: ", paste(failed$name, failed$period, collapse = ", "))
  }
  pig_periods <- pig_periods %>% filter(!map_lgl(ud, is.null))

  saveRDS(pig_periods, file)
  beep(sound = 2)
  pig_periods
}

# the number of days of each pig in each period, including those that
# are removed for having fewer than min_days
period_days <- function(pigs_nested, period_fun) {
  pigs_nested %>%
    mutate(periods = map(telemetry, ~ split_telemetry(.x, period_fun))) %>%
    dplyr::select(name, periods) %>%
    unnest(periods) %>%
    dplyr::select(name, period, start, n_fixes, n_days)
}

# 50% and 95% contours (with CIs) of each pig-period as sf (in EPSG:3112)
period_contours <- function(pig_periods, levels = contour_levels) {

  # periods ordered by when they start
  period_levels <- pig_periods %>%
    summarise(start = min(start), .by = period) %>%
    arrange(start) %>%
    pull(period)

  pig_periods %>%
    dplyr::select(name, period, ud) %>%
    crossing(level_ud = levels) %>%
    mutate(hr_sf = map2(ud, level_ud, ~ as.sf(.x, level.UD = .y, level = 0.95) %>%
                          st_transform(crs = 3112))) %>%
    dplyr::select(name, period, level_ud, hr_sf) %>%
    # as.sf already has a 'name' column (e.g. "Betty 95% est"), so rename it
    mutate(hr_sf = map(hr_sf, ~ rename(.x, contour = name))) %>%
    unnest(hr_sf) %>%
    st_as_sf(crs = 3112) %>%
    mutate(ci = word(contour, -1),
           period = factor(period, levels = period_levels),
           area_km2 = as.numeric(st_area(.)) / 1e6) %>%
    dplyr::select(name, period, level_ud, ci, area_km2)
}

month_slider_js <- r"--[
function(el, x, d) {
  var map = this;
  var months = [].concat(d.months), labels = [].concat(d.labels);
  var state = { i: 0, pts: true, timer: null };

  // the layerManager isn't always attached yet when onRender runs, so look
  // it up each time (and the first update is deferred below)
  function setGroup(g, on) {
    var mgr = map.layerManager || HTMLWidgets.find('#' + el.id).getMap().layerManager;
    var lg = mgr.getLayerGroup(g, false);
    if (!lg) return;
    if (on) map.addLayer(lg); else map.removeLayer(lg);
  }

  function update() {
    months.forEach(function(m, j) {
      setGroup(m, j === state.i);
      setGroup(m + '_fixes', j === state.i && state.pts);
    });
    lab.innerHTML = labels[state.i];
    slider.value = state.i;
  }

  var ctrl = L.control({ position: 'topright' });
  var div, slider, lab, play, chk;
  ctrl.onAdd = function() {

  lab = div.querySelector('.ms-lab');
  play = div.querySelector('.ms-play');
  chk = div.querySelector('.ms-chk');
  slider.max = months.length - 1;

  slider.addEventListener('input', function() { state.i = +slider.value; update(); });
  chk.addEventListener('change', function() { state.pts = chk.checked; update(); });

# map of monthly contours (point estimates) with a month slider
monthly_hr_map <- function(hr_sf, pts_sf, title, interval_ms = 1500) {

  hr_sf <- hr_sf %>%
    # filter(ci == "est") %>%
    st_transform(crs = 4326) %>%
    mutate(popup = paste0("ID: ", name,
                          "<br>Month: ", format(month_date, "%b %Y"),
                          "<br>Contour: ", level_ud * 100, "%",
                          "<br>Area: ", round(area_km2, 2), " km<sup>2</sup>"))

  months <- sort(unique(hr_sf$year_month))
  pal <- colorFactor(viridisLite::viridis(n_distinct(pig_all$name), option = "H"),
                     domain = sort(unique(pig_all$name)))

  m <- leaflet(options = leafletOptions(preferCanvas = TRUE)) %>%
    addProviderTiles(providers$Esri.WorldImagery)

  for (mon in months) {
    hr_95 <- hr_sf %>% filter(year_month == mon, level_ud == 0.95)
    hr_50 <- hr_sf %>% filter(year_month == mon, level_ud == 0.5)
    pts <- pts_sf %>% filter(year_month == mon)

    m <- m %>%
      addCircleMarkers(data = pts, radius = 2, stroke = FALSE,
                       fillColor = ~ pal(name), fillOpacity = 0.6,
                       group = paste0(mon, "_fixes")) %>%
      addPolygons(data = hr_95, color = ~ pal(name), weight = 2, opacity = 1,
                  fillColor = ~ pal(name), fillOpacity = 0.1,
                  popup = ~ popup, group = mon) %>%
      addPolygons(data = hr_50, color = ~ pal(name), weight = 1.5, opacity = 1,
                  dashArray = "4 4",
                  fillColor = ~ pal(name), fillOpacity = 0.3,
                  popup = ~ popup, group = mon)
  }

  m %>%
    addLegend(position = "bottomright", pal = pal,
              values = sort(unique(hr_sf$name)), title = "Pig ID") %>%
    addScaleBar(position = "bottomleft",
                options = scaleBarOptions(imperial = FALSE)) %>%
    htmlwidgets::onRender(month_slider_js,
                          data = list(months = months,
                                      labels = format(ym(months), "%B %Y"),
                                      title = title,
                                      interval = interval_ms))
}
