Abstract

Investigating how pig movement and habitat use patterns change with weather variables (temperature, rainfall, solar exposure) in the East Alligator region of Kakadu National Park.

Daily distance travelled

Show code 
start_date <- min(pig_all_steps$date)
bin_width <- 3 # days

# daily distance travelled for each pig
daily_distance_id <- pig_all_steps %>%
  group_by(name, date) %>%
  summarise(
    sum_sl = sum(sl_, na.rm = TRUE),
    .groups = "drop"
  )

# get the rolling mean of the daily distance travelled for each pig
daily_distance_id_rolling_mean <- daily_distance_id %>%
  mutate(
    bin_num  = as.numeric(date - start_date) %/% bin_width,
    date_bin = start_date + bin_num * bin_width + (bin_width-1)/2      # +2 = bin midpoint (5 days: 0-4)
  ) %>%
  group_by(name, date_bin) %>%
  summarise(
    id_sum_sl = mean(sum_sl, na.rm = TRUE),
    q25 = quantile(sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(sum_sl, probs = 0.75, na.rm = TRUE),
    n   = sum(!is.na(sum_sl)),
    .groups = "drop"
  )

# average daily distance travelled across all pigs
daily_distance_all <- daily_distance_id_rolling_mean %>%
  group_by(date_bin) %>%
  summarise(
    mean_sum_sl = mean(id_sum_sl, na.rm = TRUE),
    q25 = quantile(id_sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(id_sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(id_sum_sl, probs = 0.75, na.rm = TRUE),
    n   = n(),
    .groups = "drop"
  )

# bin a weather variable into quantile bins so each bin has a similar number of fixes
bin_weather <- function(x, n_bins = 10) {
  # categorical or flag variables (cloud amounts, sky clear flag) are used as they are
  if (!is.numeric(x) || n_distinct(x, na.rm = TRUE) <= n_bins) return(as.factor(x))
  # variables that are mostly zero (rain, calm wind) get a bin of their own for zeros,
  # and quantile bins for the non-zero values
  zero_bin <- mean(x == 0, na.rm = TRUE) > 1 / n_bins
  x_to_bin <- if (zero_bin) x[x > 0] else x
  # rounded so floating point error doesn't create near-duplicate breaks
  breaks <- unique(signif(quantile(x_to_bin, probs = seq(0, 1, length.out = n_bins + 1), na.rm = TRUE), 6))
  bins <- cut(x, breaks = breaks, include.lowest = TRUE, dig.lab = 4)
  if (zero_bin) {
    bins <- fct_expand(bins, "0") %>% replace(x == 0, "0") %>% fct_relevel("0")
  }
  bins
}

## Daily distance travelled

```{r}
start_date <- min(pig_all_steps$date)
bin_width <- 3 # days

# daily distance travelled for each pig
daily_distance_id <- pig_all_steps %>%
  group_by(name, date) %>%
  summarise(
    sum_sl = sum(sl_, na.rm = TRUE),
    .groups = "drop"
  )

# get the rolling mean of the daily distance travelled for each pig
daily_distance_id_rolling_mean <- daily_distance_id %>%
  mutate(
    bin_num  = as.numeric(date - start_date) %/% bin_width,
    date_bin = start_date + bin_num * bin_width + (bin_width-1)/2      # +2 = bin midpoint (5 days: 0-4)
  ) %>%
  group_by(name, date_bin) %>%
  summarise(
    id_sum_sl = mean(sum_sl, na.rm = TRUE),
    q25 = quantile(sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(sum_sl, probs = 0.75, na.rm = TRUE),
    n   = sum(!is.na(sum_sl)),
    .groups = "drop"
  )

# average daily distance travelled across all pigs
daily_distance_all <- daily_distance_id_rolling_mean %>%
  group_by(date_bin) %>%
  summarise(
    mean_sum_sl = mean(id_sum_sl, na.rm = TRUE),
    q25 = quantile(id_sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(id_sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(id_sum_sl, probs = 0.75, na.rm = TRUE),
    n   = n(),
    .groups = "drop"
  )
```

```{r}
# bin a weather variable into quantile bins so each bin has a similar number of fixes
bin_weather <- function(x, n_bins = 10) {
  # categorical or flag variables (cloud amounts, sky clear flag) are used as they are
  if (!is.numeric(x) || n_distinct(x, na.rm = TRUE) <= n_bins) return(as.factor(x))
  # variables that are mostly zero (rain, calm wind) get a bin of their own for zeros,
  # and quantile bins for the non-zero values
  zero_bin <- mean(x == 0, na.rm = TRUE) > 1 / n_bins
  x_to_bin <- if (zero_bin) x[x > 0] else x
  # rounded so floating point error doesn't create near-duplicate breaks
  breaks <- unique(signif(quantile(x_to_bin, probs = seq(0, 1, length.out = n_bins + 1), na.rm = TRUE), 6))
  bins <- cut(x, breaks = breaks, include.lowest = TRUE, dig.lab = 4)
  if (zero_bin) {
    bins <- fct_expand(bins, "0") %>% replace(x == 0, "0") %>% fct_relevel("0")
  }
  bins
}
```

## Max daily temp

```{r}
#| results: asis
#| fig-width: 6
#| fig-height: 4

cat("::: {.panel-tabset}\n\n")

for (veg_col in veg_cols) {

  veg_label <- gsub("_", " ", sub("^veg_", "", veg_col)) %>%
    stringr::str_to_sentence()

  bin_summary <- pig_all_steps %>%
    group_by(day_frac_bin, max_temp_bin) %>%
    summarise(
      n = sum(!is.na(.data[[veg_col]])),
      k = sum(.data[[veg_col]], na.rm = TRUE),
      .groups = "drop"
    ) %>%
    filter(n > 50) %>% # how many fixes per bin? (to avoid bins with very few fixes)
    mutate(
      p      = k / n,
      centre = (p + z^2 / (2 * n)) / (1 + z^2 / n),
      half   = z * sqrt(p * (1 - p) / n + z^2 / (4 * n^2)) / (1 + z^2 / n),
      lower  = pmax(0, centre - half),
      upper  = pmin(1, centre + half)
    )

  p <- ggplot(bin_summary, 
              aes(x = day_frac_bin, 
                  colour = as.factor(max_temp_bin), fill = as.factor(max_temp_bin))) +
    geom_ribbon(aes(ymin = lower, ymax = upper),
                alpha = 0.25, colour = alpha("black", 0.75), linewidth = 0.15) +
    geom_line(aes(y = p)) +
    geom_point(aes(y = p), size = 1) +
    
    # add lines for sunrise and sunset (day_fraction = 0 and 1)
    geom_vline(xintercept = c(0,1), colour = "black", linetype = "dashed") +
    annotate("text", x = c(0, 1), y = Inf, 
             label = c("Sunrise", "Sunset"),
             hjust = -0.1, vjust = 2, size = 3.5) +
    
    scale_y_continuous("Proportion of fixes",
                       limits = c(0, 1), expand = expansion(mult = c(0, 0.05))) +
    scale_x_continuous("Day fraction", 
                       limits = c(-0.5, 1.36), breaks = seq(-1, 1.5, by = 0.2)) +
    scale_colour_brewer(name = "Max daily\ntemp (°C)", palette = "Reds") +
    scale_fill_brewer(name = "Max daily\ntemp (°C)", palette = "Reds") +
    labs(title = veg_label,
         # subtitle = "Daily usage across max_temp_bins"
         ) +
    theme_bw() +
    theme(legend.position = "right")

  ggsave(file.path(outputs_veg_dir, paste0(veg_col, "_max_temp_bins.png")),
         plot = p, width = 150, height = 90, scale = 1.25, units = "mm", dpi = 600)

  # one tab per habitat: heading first, then the plot
  cat("\n\n#### ", veg_label, "\n\n")
  print(p)
}
```

# Vegetation responses to half-hourly weather variables

These analyses look at how the pigs selection of vegetation classes changes due to real-time to weather conditions.

```{r}
#| results: asis
#| fig-width: 9
#| fig-height: 7

# weather variables are binned with bin_weather(), defined in the step length median and quantiles chunk

# one tab per weather variable, with a facet per vegetation class
cat("::: {.panel-tabset}\n\n")

for (var in halfhourly_weather_vars) {

  is_continuous <- is.numeric(pig_all_steps[[var]]) &&
    n_distinct(pig_all_steps[[var]], na.rm = TRUE) > 10

  bin_summary <- pig_all_steps %>%
    mutate(weather_bin = bin_weather(.data[[var]])) %>%
    filter(!is.na(weather_bin)) %>%
    select(all_of(var), weather_bin, all_of(veg_cols)) %>%
    pivot_longer(all_of(veg_cols), names_to = "veg_col", values_to = "in_veg") %>%
    group_by(veg_col, weather_bin) %>%
    summarise(
      # plot continuous variables at the median value of each bin
      weather_x = if (is_continuous) median(.data[[var]]) else NA,
      n = sum(!is.na(in_veg)),
      k = sum(in_veg, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    filter(n > 50) %>% # how many fixes per bin? (to avoid bins with very few fixes)
    mutate(
      p      = k / n,
      centre = (p + z^2 / (2 * n)) / (1 + z^2 / n),
      half   = z * sqrt(p * (1 - p) / n + z^2 / (4 * n^2)) / (1 + z^2 / n),
      lower  = pmax(0, centre - half),
      upper  = pmin(1, centre + half),
      veg_label = gsub("_", " ", sub("^veg_", "", veg_col)) %>% stringr::str_to_sentence()
    )

  # continuous variables on a continuous axis, categories on a discrete axis
  if (is_continuous) {
    p <- ggplot(bin_summary, aes(x = weather_x))
  } else {
    p <- ggplot(bin_summary, aes(x = weather_bin))
  }

  p <- p +
    geom_ribbon(aes(ymin = lower, ymax = upper, group = 1),
                alpha = 0.25, colour = alpha("black", 0.75), linewidth = 0.15) +
    geom_line(aes(y = p, group = 1)) +
    geom_point(aes(y = p), size = 1) +
    facet_wrap(~ veg_label, scales = "free_y") +
    scale_y_continuous("Proportion of fixes") +
    coord_cartesian(ylim = c(0, 1)) +
    labs(title = weather_label(var), x = weather_label(var)) +
    theme_bw()

  # one tab per weather variable: heading first, then the plot
  cat("\n\n## ", var, "\n\n")
  print(p)
}

cat("\n\n:::\n")
```
