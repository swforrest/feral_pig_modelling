Abstract

Investigating how pig movement and habitat use patterns change with weather variables (temperature, rainfall, solar exposure) in the East Alligator region of Kakadu National Park.

Step lengths by the month of the year

Show code 
bin_width <- 0.5  # hours

bin_means <- pig_all_steps %>%
  mutate(bin = floor(hour / bin_width) * bin_width + bin_width / 2) %>%
  group_by(bin, month) %>%
  summarise(
    mean_sl = mean(sl_, na.rm = TRUE),
    se      = sd(sl_, na.rm = TRUE) / sqrt(sum(!is.na(sl_))),
    q25     = quantile(sl_, probs = 0.25, na.rm = TRUE),
    q50     = quantile(sl_, probs = 0.5, na.rm = TRUE),
    q75     = quantile(sl_, probs = 0.75, na.rm = TRUE),
    n       = sum(!is.na(sl_)),
    .groups = "drop"

             colour = as.factor(month), fill = as.factor(month))) +

Daily distance travelled

Show code 
start_date <- min(pig_all_steps$date)
bin_width <- 3 # days

# daily distance travelled for each pig
daily_distance_id <- pig_all_steps %>%
  group_by(name, date) %>%
  summarise(
    sum_sl = sum(sl_, na.rm = TRUE),
    .groups = "drop"
  )

# get the rolling mean of the daily distance travelled for each pig
daily_distance_id_rolling_mean <- daily_distance_id %>%
  mutate(
    bin_num  = as.numeric(date - start_date) %/% bin_width,
    date_bin = start_date + bin_num * bin_width + (bin_width-1)/2      # +2 = bin midpoint (5 days: 0-4)
  ) %>%
  group_by(name, date_bin) %>%
  summarise(
    id_sum_sl = mean(sum_sl, na.rm = TRUE),
    q25 = quantile(sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(sum_sl, probs = 0.75, na.rm = TRUE),
    n   = sum(!is.na(sum_sl)),
    .groups = "drop"
  )

# average daily distance travelled across all pigs
daily_distance_all <- daily_distance_id_rolling_mean %>%
  group_by(date_bin) %>%
  summarise(
    mean_sum_sl = mean(id_sum_sl, na.rm = TRUE),
    q25 = quantile(id_sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(id_sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(id_sum_sl, probs = 0.75, na.rm = TRUE),
    n   = n(),
    .groups = "drop"
  )

## Daily distance travelled

```{r}
start_date <- min(pig_all_steps$date)
bin_width <- 3 # days

# daily distance travelled for each pig
daily_distance_id <- pig_all_steps %>%
  group_by(name, date) %>%
  summarise(
    sum_sl = sum(sl_, na.rm = TRUE),
    .groups = "drop"
  )

# get the rolling mean of the daily distance travelled for each pig
daily_distance_id_rolling_mean <- daily_distance_id %>%
  mutate(
    bin_num  = as.numeric(date - start_date) %/% bin_width,
    date_bin = start_date + bin_num * bin_width + (bin_width-1)/2      # +2 = bin midpoint (5 days: 0-4)
  ) %>%
  group_by(name, date_bin) %>%
  summarise(
    id_sum_sl = mean(sum_sl, na.rm = TRUE),
    q25 = quantile(sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(sum_sl, probs = 0.75, na.rm = TRUE),
    n   = sum(!is.na(sum_sl)),
    .groups = "drop"
  )

# average daily distance travelled across all pigs
daily_distance_all <- daily_distance_id_rolling_mean %>%
  group_by(date_bin) %>%
  summarise(
    mean_sum_sl = mean(id_sum_sl, na.rm = TRUE),
    q25 = quantile(id_sum_sl, probs = 0.25, na.rm = TRUE),
    q50 = quantile(id_sum_sl, probs = 0.5,  na.rm = TRUE),
    q75 = quantile(id_sum_sl, probs = 0.75, na.rm = TRUE),
    n   = n(),
    .groups = "drop"
  )

```

## Months of the year

At a finer scale, what about for the different months of the year, for each of the vegetation classes? Here we look at the proportion of fixes in different vegetation types across the day fraction (where 0 = sunrise and 1 = sunset) for each month of the year.

```{r}
#| results: asis
#| fig-width: 6
#| fig-height: 4

veg_cols <- grep("^veg_", names(pig_all_steps), value = TRUE)
veg_cols <- veg_cols[veg_cols != "veg_water"] # drop the water category as it isn't used

cat("::: {.panel-tabset}\n\n")

for (veg_col in veg_cols) {

  veg_label <- gsub("_", " ", sub("^veg_", "", veg_col)) %>%
    stringr::str_to_sentence()

  bin_summary <- pig_all_steps %>%
    group_by(day_frac_bin, month) %>%
    summarise(
      n = sum(!is.na(.data[[veg_col]])),
      k = sum(.data[[veg_col]], na.rm = TRUE),
      .groups = "drop"
    ) %>%
    filter(n > 50) %>% # how many fixes per bin? (to avoid bins with very few fixes)
    mutate(
      p      = k / n,
      centre = (p + z^2 / (2 * n)) / (1 + z^2 / n),
      half   = z * sqrt(p * (1 - p) / n + z^2 / (4 * n^2)) / (1 + z^2 / n),
      lower  = pmax(0, centre - half),
      upper  = pmin(1, centre + half)
```
